# Only executable entry points live here; shared functionality stays in src/include.
if(TINYENGINE_BUILD_TESTS)
    enable_testing()
endif()
add_subdirectory(project_tool)

add_executable(tinyengine_asset_tool asset_tool/asset_tool.cpp)
target_link_libraries(tinyengine_asset_tool PRIVATE tinyengine)
if(TINYENGINE_BUILD_IMPORTERS)
    target_link_libraries(tinyengine_asset_tool PRIVATE tinyengine_importers)
    target_compile_definitions(tinyengine_asset_tool PRIVATE TINYENGINE_HAS_ASSIMP=1)
endif()
add_executable(tinyengine_scene_tool scene_tool/scene_tool.cpp)
target_link_libraries(tinyengine_scene_tool PRIVATE tinyengine)
add_executable(tinyengine_package_tool package_tool/package_tool.cpp)
target_link_libraries(tinyengine_package_tool PRIVATE tinyengine)
add_executable(tinyengine_commandlet commandlet/commandlet.cpp)
target_link_libraries(tinyengine_commandlet PRIVATE tinyengine)
foreach(target tinyengine_asset_tool tinyengine_scene_tool tinyengine_package_tool tinyengine_commandlet)
    set_target_properties(${target} PROPERTIES FOLDER programs RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endforeach()
