cmake_minimum_required(VERSION 3.21)
project(TinySDKConsumer LANGUAGES CXX)
find_package(TinyEngine CONFIG REQUIRED)
get_target_property(imported_links TinyEngine::ProjectSDK INTERFACE_LINK_LIBRARIES)
if(imported_links MATCHES "SDL|imgui|(^|;)tinyengine($|;)")
    message(FATAL_ERROR "ProjectSDK must not depend on the full engine")
endif()
if(DEFINED EXPECTED_SDK_PREFIX)
get_target_property(imported_includes TinyEngine::ProjectSDK INTERFACE_INCLUDE_DIRECTORIES)
foreach(config DEBUG RELEASE)
    get_target_property(location TinyEngine::ProjectSDK IMPORTED_LOCATION_${config})
    if(NOT EXISTS "${location}" OR NOT location MATCHES "^${EXPECTED_SDK_PREFIX}/")
        message(FATAL_ERROR "Missing or non-relocatable ${config} SDK library: ${location}")
    endif()
endforeach()
foreach(path IN LISTS imported_includes)
    if(NOT path MATCHES "^${EXPECTED_SDK_PREFIX}/")
        message(FATAL_ERROR "ProjectSDK contains a non-relocatable include: ${path}")
    endif()
endforeach()
endif()
tinyengine_add_project_module(TARGET RotateGame PROJECT_ID sdk-consumer MODULE_ID game.rotate
    PROJECT_FILE Game.tinyproject SOURCES Source/Game.cpp)
