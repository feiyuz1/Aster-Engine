cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(TinyEngineProjectTool LANGUAGES CXX)
    include(CTest)
endif()
if(NOT WIN32)
    message(STATUS "ProjectTool shell integration is available on Windows only")
    return()
endif()
get_filename_component(TINYENGINE_TOOL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(tinyengine_project_tool_services STATIC
    "${TINYENGINE_TOOL_ROOT}/src/Project/SDKPaths.cpp"
    "${TINYENGINE_TOOL_ROOT}/src/Project/SolutionGenerator.cpp"
    ShellMenu.cpp)
target_include_directories(tinyengine_project_tool_services PUBLIC "${TINYENGINE_TOOL_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(tinyengine_project_tool_services PUBLIC cxx_std_17)
target_compile_definitions(tinyengine_project_tool_services PRIVATE TINYENGINE_CMAKE_EXECUTABLE="${CMAKE_COMMAND}")
target_link_libraries(tinyengine_project_tool_services PUBLIC advapi32 shell32)
add_executable(tinyengine_project_tool WIN32 Main.cpp)
target_link_libraries(tinyengine_project_tool PRIVATE tinyengine_project_tool_services user32)
set_target_properties(tinyengine_project_tool PROPERTIES OUTPUT_NAME TinyEngineProjectTool FOLDER Programs)
if(MSVC)
    foreach(target tinyengine_project_tool_services tinyengine_project_tool)
        target_compile_options(${target} PRIVATE /utf-8 /FS)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
endif()
install(TARGETS tinyengine_project_tool RUNTIME DESTINATION Programs/ProjectTool COMPONENT ProjectTool)
if(BUILD_TESTING OR TINYENGINE_BUILD_TESTS)
    enable_testing()
    add_executable(tinyengine_project_shell_menu_tests "${TINYENGINE_TOOL_ROOT}/tests/ProjectShellMenuTests.cpp")
    target_link_libraries(tinyengine_project_shell_menu_tests PRIVATE tinyengine_project_tool_services)
    if(MSVC)
        target_compile_options(tinyengine_project_shell_menu_tests PRIVATE /utf-8 /FS)
        set_property(TARGET tinyengine_project_shell_menu_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_test(NAME tinyengine_project_shell_menu COMMAND tinyengine_project_shell_menu_tests)
    add_test(NAME tinyengine_project_tool_cli COMMAND "${CMAKE_COMMAND}" "-DTOOL=$<TARGET_FILE:tinyengine_project_tool>"
        -P "${TINYENGINE_TOOL_ROOT}/tests/ProjectToolTests.cmake")
endif()
